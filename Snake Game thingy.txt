
import pygame
import random
import sys

# Initialize pygame
pygame.init()

# Colors
WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
RED = (213, 50, 80)
GREEN = (0, 255, 0)

# Screen settings
WIDTH = 600
HEIGHT = 600
BLOCK_SIZE = 20

screen = pygame.display.set_mode((WIDTH, HEIGHT))
pygame.display.set_caption('Snake Game by Grok')

clock = pygame.time.Clock()

# Fonts
font_style = pygame.font.SysFont("bahnschrift", 25)
score_font = pygame.font.SysFont("comicsansms", 35)


def show_score(score):
    value = score_font.render(f"Score: {score}", True, WHITE)
    screen.blit(value, [10, 10])


def game_over_text():
    text = font_style.render("Game Over! Press Q - Quit or C - Play Again", True, RED)
    screen.blit(text, [WIDTH//2 - 280, HEIGHT//2])


def draw_snake(snake_list):
    for x, y in snake_list:
        pygame.draw.rect(screen, GREEN, [x, y, BLOCK_SIZE, BLOCK_SIZE])


def game_loop():
    # Game variables
    game_over = False
    game_close = False

    x = WIDTH // 2
    y = HEIGHT // 2

    x_change = 0
    y_change = 0

    snake_list = []
    snake_length = 1

    # Starting food position
    food_x = random.randrange(0, WIDTH // BLOCK_SIZE) * BLOCK_SIZE
    food_y = random.randrange(0, HEIGHT // BLOCK_SIZE) * BLOCK_SIZE

    score = 0

    while not game_over:

        # Game Over Screen
        while game_close:
            screen.fill(BLACK)
            game_over_text()
            show_score(score)
            pygame.display.update()

            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    pygame.quit()
                    sys.exit()
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_q:
                        pygame.quit()
                        sys.exit()
                    if event.key == pygame.K_c:
                        return  # Restart the game

        # Event handling (keyboard)
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                sys.exit()

            if event.type == pygame.KEYDOWN:
                if event.key in (pygame.K_LEFT, pygame.K_a) and x_change != BLOCK_SIZE:
                    x_change = -BLOCK_SIZE
                    y_change = 0
                elif event.key in (pygame.K_RIGHT, pygame.K_d) and x_change != -BLOCK_SIZE:
                    x_change = BLOCK_SIZE
                    y_change = 0
                elif event.key in (pygame.K_UP, pygame.K_w) and y_change != BLOCK_SIZE:
                    y_change = -BLOCK_SIZE
                    x_change = 0
                elif event.key in (pygame.K_DOWN, pygame.K_s) and y_change != -BLOCK_SIZE:
                    y_change = BLOCK_SIZE
                    x_change = 0

        # Move the snake
        x += x_change
        y += y_change

        # Check wall collision
        if x >= WIDTH or x < 0 or y >= HEIGHT or y < 0:
            game_close = True
            continue

        screen.fill(BLACK)

        # Draw food
        pygame.draw.rect(screen, RED, [food_x, food_y, BLOCK_SIZE, BLOCK_SIZE])

        # Add new head to snake
        snake_head = [x, y]
        snake_list.append(snake_head)

        # Keep snake at correct length
        if len(snake_list) > snake_length:
            del snake_list[0]

        # Check self collision
        if snake_head in snake_list[:-1]:
            game_close = True

        draw_snake(snake_list)
        show_score(score)
        pygame.display.update()

        # Eating food
        if x == food_x and y == food_y:
            food_x = random.randrange(0, WIDTH // BLOCK_SIZE) * BLOCK_SIZE
            food_y = random.randrange(0, HEIGHT // BLOCK_SIZE) * BLOCK_SIZE
            snake_length += 1
            score += 10

        clock.tick(12)  # Game speed (higher = faster)


# Start the game
if __name__ == "__main__":
    while True:
        game_loop()
